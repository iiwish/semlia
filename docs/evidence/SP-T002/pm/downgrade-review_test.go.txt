package db_test

import (
	"context"
	"strings"
	"testing"
)

func TestPMCoverageAttemptBlocksDowngrade(t *testing.T) {
	pool, migrator := semanticProductionDatabase(t, "source22_pm_attempt", 22)
	workspace, source, _, run := seedSourceHistoryRun(t, pool)
	ctx := context.Background()
	if _, err := pool.Exec(ctx, `UPDATE discovery_runs SET status='running',completed_at=NULL WHERE id=$1`, run.UUID()); err != nil {
		t.Fatal(err)
	}
	if _, err := pool.Exec(ctx, `INSERT INTO source_coverage_heads(workspace_id,source_connection_id,coverage_key,selector_digest,latest_attempt_run_id,latest_attempt_status,version) VALUES($1,$2,'catalog',$3,$4,'partial',1)`, workspace.UUID(), source.UUID(), distributionDigest("selector"), run.UUID()); err != nil {
		t.Fatal(err)
	}
	var heads, snapshots int
	if err := pool.QueryRow(ctx, `SELECT (SELECT count(*) FROM source_coverage_heads),(SELECT count(*) FROM source_snapshots)`).Scan(&heads, &snapshots); err != nil {
		t.Fatal(err)
	}
	t.Logf("before down: coverageHeads=%d snapshots=%d", heads, snapshots)
	err := migrator.Steps(-1)
	var exists bool
	if scanErr := pool.QueryRow(ctx, `SELECT to_regclass('source_coverage_heads') IS NOT NULL`).Scan(&exists); scanErr != nil {
		t.Fatal(scanErr)
	}
	t.Logf("after down: error=%v coverageHeadsTableExists=%v", err, exists)
	if err == nil || !strings.Contains(err.Error(), "DOWN_MIGRATION_UNSAFE") || !exists {
		t.Fatal("downgrade removed an authoritative coverage attempt")
	}
}
