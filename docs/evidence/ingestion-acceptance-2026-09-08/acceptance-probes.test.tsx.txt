describe("acceptance counterexamples", () => {
  it("does not create a second proposal after closing an association failure", async () => {
    const user = userEvent.setup();
    mocks.createAndSubmitProposal.mockResolvedValue({ id: "prp_first" });
    mocks.decideSemanticCandidate.mockRejectedValue(new Error("association unavailable"));
    render(<LiveSourcesView focusIndex={2} runDetailBackRequestEpoch={0} onRunDetailOpenChange={() => {}} onOpenProposal={() => {}} />);
    await user.click(await screen.findByRole("tab", { name: /语义候选/ }));
    const submit = async () => {
      await user.click(screen.getByRole("button", { name: `查看语义候选 ${candidate.title}` }));
      await user.selectOptions(screen.getByLabelText("目标语义资产"), assets[0].id);
      await user.clear(screen.getByLabelText("候选定义"));
      await user.type(screen.getByLabelText("候选定义"), "Reviewed definition");
      await user.type(screen.getByLabelText("候选决策依据"), "Reviewed fields");
      await user.click(screen.getByRole("button", { name: "创建并提交提案" }));
      await screen.findByRole("alert");
    };
    await submit();
    await user.click(screen.getByRole("button", { name: `关闭${candidate.title}` }));
    await submit();
    expect(mocks.createAndSubmitProposal).toHaveBeenCalledTimes(1);
  });
  it("refreshes candidates when a newly scheduled run is opened completed", async () => {
    const user = userEvent.setup();
    const newRunId = "run_new";
    const newRevision = "srv_new";
    const schedule = { id: "sch_test", sourceId: source.id, expression: "0 2 * * *", timezone: "Asia/Shanghai", enabled: true, version: 1, misfirePolicy: "skip", createdAt: source.createdAt, updatedAt: source.updatedAt };
    mocks.listSchedules.mockResolvedValue({ items: [schedule], total: 1, limit: 50 });
    mocks.listOccurrences.mockResolvedValue({ items: [{ id: "occ_test", scheduleId: schedule.id, sourceId: source.id, eligibleAt: run.createdAt, triggerKind: "cron", state: "enqueued", discoveryRunId: newRunId, runtimeRunId: newRunId }], total: 1, limit: 50 });
    mocks.getDiscoveryRun.mockResolvedValue({ ...run, id: newRunId, status: "succeeded", sourceRevisionId: newRevision, adapterVersion: "v1", findings: [] });
    render(<LiveSourcesView focusIndex={1} runDetailBackRequestEpoch={0} onRunDetailOpenChange={() => {}} onOpenProposal={() => {}} />);
    await user.click(await screen.findByRole("button", { name: "查看计划 sch_test" }));
    mocks.listSemanticCandidates.mockResolvedValue({ items: [{ ...candidate, id: "scd_new", title: "new.results", discoveryRunId: newRunId, sourceRevisionId: newRevision }], total: 1, limit: 50 });
    await user.click(await screen.findByRole("button", { name: "打开运行" }));
    expect(await screen.findByRole("button", { name: /new.results/ })).toBeVisible();
  });
});
